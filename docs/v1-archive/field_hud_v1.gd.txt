# 원형 스윕 게이지. 남은 비율만 링 + 옅은 부채꼴로 그린다.
class CycleSweepGauge:
	extends Control

	var ratio := 1.0
	var radius := 33.0
	var thickness := 5.0
	var ring_color := Color("e78a45")
	var track_color := Color("232c3d")

	func _draw() -> void:
		var center := size * 0.5
		draw_arc(center, radius, 0.0, TAU, 48, track_color, thickness, false)
		var visible_ratio := clampf(ratio, 0.0, 1.0)
		if visible_ratio <= 0.004:
			return
		var start := -PI * 0.5
		var end := start + TAU * visible_ratio
		var steps := maxi(3, int(ceil(48.0 * visible_ratio)))
		var sector := PackedVector2Array()
		sector.append(center)
		for step in steps + 1:
			sector.append(center + Vector2.from_angle(lerpf(start, end, float(step) / float(steps))) * radius)
		draw_colored_polygon(sector, Color(ring_color, 0.07))
		draw_arc(center, radius, start, end, 48, Color(ring_color, 0.88), thickness, false)
		# 12시 기준점. 되감기가 어디서 끝나는지 눈으로 고정해 준다.
		draw_rect(Rect2(center + Vector2(-1.0, -radius - thickness), Vector2(2.0, thickness * 2.0)), ring_color, true)

func _update_cycle_rewind() -> void:
	if not is_instance_valid(cycle_rewind_reel) or not is_instance_valid(cycle_hud_panel) or factory == null:
		return
	var rewinding := cycle_hud_panel.visible and not inside_castle and is_instance_valid(player_cycle) and player_cycle.reloading and not factory.slots.is_empty()
	if rewinding != cycle_rewind_active:
		cycle_rewind_active = rewinding
		_set_cycle_rewind_visible(rewinding)
		if not rewinding:
			# 되감기가 끝난 프레임에 바로 칸 01 상태로 이어지도록 즉시 갱신합니다.
			_update_deal_cycle_hud()
	if not rewinding:
		return
	var slot_count := maxi(1, factory.slots.size())
	var phase := clampf(player_cycle.progress_ratio(), 0.0, 1.0)
	# 뒤로 갈수록 느려지는 감속. 칩이 적을 때도 최소 3칸 이상 흘러가도록 한 바퀴를 더 돈다.
	var loops := maxi(1, ceili(4.0 / float(slot_count)))
	var travel := float(slot_count * loops) - 1.0
	var eased := 1.0 - pow(1.0 - phase, 2.4)
	var virtual_slot := (1.0 - eased) * travel
	var base := floori(virtual_slot)
	var fraction := virtual_slot - float(base)
	var offset := -fraction * CYCLE_REWIND_PITCH * CYCLE_REWIND_TRAVEL
	var half := int((CYCLE_REWIND_CHIPS - 1) / 2)
	for chip_index in cycle_rewind_chips.size():
		var chip := cycle_rewind_chips[chip_index]
		if not is_instance_valid(chip):
			continue
		var column := chip_index - half
		var center_x := CYCLE_REWIND_PLAYHEAD_X + float(column) * CYCLE_REWIND_PITCH + offset
		var closeness := clampf(1.0 - absf(center_x - CYCLE_REWIND_PLAYHEAD_X) / (CYCLE_REWIND_PITCH * CYCLE_REWIND_FADE_SPAN), 0.0, 1.0)
		chip.position = Vector2(center_x - cycle_rewind_reel.position.x - CYCLE_REWIND_CHIP_SIDE * 0.5, CYCLE_HUD_LINE_Y - cycle_rewind_reel.position.y - CYCLE_REWIND_CHIP_SIDE * 0.5)
		chip.scale = Vector2.ONE * lerpf(0.7, 1.0, closeness)
		chip.modulate = Color(1.0, 1.0, 1.0, pow(closeness, 0.85))
		chip.rotation = deg_to_rad(lerpf(7.0, 0.0, closeness) * signf(center_x - CYCLE_REWIND_PLAYHEAD_X))
		_apply_cycle_rewind_chip(chip, wrapi(base + column, 0, slot_count))
	cycle_rewind_gauge.ratio = 1.0 - phase
	cycle_rewind_gauge.queue_redraw()
	cycle_rewind_time_text.text = "%.2f초" % player_cycle.reload_remaining
	cycle_rewind_bar_fill.size.x = (CYCLE_REWIND_INFO_W - 2.0) * (1.0 - phase)
	# 릴이 빠를 때만 잔상이 진해집니다(감속과 함께 자연히 사라짐).
	cycle_rewind_streak.color = Color(GamePalette.ORANGE, clampf(pow(1.0 - phase, 1.6) * 0.16, 0.0, 0.16))

func _apply_cycle_rewind_chip(chip: Panel, slot_index: int) -> void:
	var card := _cycle_slot_lead_card(slot_index)
	var color := GamePalette.STONE_LIGHT if card.is_empty() else _factory_card_color(card)
	var icon := chip.get_node_or_null("Icon") as PixelSkillIcon
	if is_instance_valid(icon):
		var visual_id := "basic" if card.is_empty() else String(card.get("id", "basic"))
		if icon.skill_id != visual_id or icon.icon_color != color:
			icon.setup(visual_id, color, false)
	var number := chip.get_node_or_null("Number") as Label
	if is_instance_valid(number):
		number.text = "칸 %02d" % (slot_index + 1)

func _cycle_slot_lead_card(slot_index: int) -> Dictionary:

	if factory == null:
		return {}

func _apply_cycle_chip(chip: Panel, slot_index: int, emphasised: bool) -> void:
	if not is_instance_valid(chip) or factory == null:
		return
	var card := _cycle_slot_lead_card(slot_index)
	# v2 뱃지는 레인 수가 아니라 그 칸에 박힌 **각인 개수**입니다(0이면 숨김).
	var rune_count := factory.rune_count_on(slot_index)
	var color := GamePalette.STONE_LIGHT if card.is_empty() else _factory_card_color(card)
	var icon := chip.get_node_or_null("Icon") as PixelSkillIcon
	var badge := chip.get_node_or_null("Badge") as Label
	if is_instance_valid(icon):
		var visual_id := String(card.get("id", "basic")) if not card.is_empty() else "basic"
		if icon.skill_id != visual_id or icon.icon_color != color:
			icon.setup(visual_id, color, false)
	if is_instance_valid(badge):
		badge.visible = rune_count > 0
		badge.text = "각%d" % rune_count
	# 스타일박스는 색·강조가 바뀔 때만 새로 만듭니다(HUD는 초당 10회 갱신됩니다).
	var style_key := "%s|%d" % [color.to_html(false), int(emphasised)]
	if String(chip.get_meta("cycle_style_key", "")) != style_key:
		chip.set_meta("cycle_style_key", style_key)
		var border := color.darkened(0.15) if emphasised else color.darkened(0.55)
		var background := color.darkened(0.74) if emphasised else color.darkened(0.86)
		chip.add_theme_stylebox_override("panel", _panel_style(background, border, 0, UI_BORDER_CARD if emphasised else UI_BORDER_FRAME))
	chip.modulate = Color(1.0, 1.0, 1.0, 1.0 if emphasised else 0.6)
