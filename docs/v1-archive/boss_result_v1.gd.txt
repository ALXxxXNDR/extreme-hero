# v2: 마왕도 5칸이다(§6.1). 배치·각인은 DemonLord가 소유하므로 여기서는 옮겨 담기만 한다.
# 버린 아이템은 레일이 아니라 마왕의 장비로 간다(§5.4 — 칸을 먹지 않는다).
func _build_boss_factory() -> FactoryDeck:
	var deck: FactoryDeck = FACTORY_SCRIPT.new()
	deck.reset(GameTuning.BOSS_SLOT_COUNT)
	demon_lord.sync_runes(rng)
	for entry: Dictionary in demon_lord.slot_layout():
		var slot_index := int(entry.get("index", 0))
		var card: Dictionary = entry.get("card", {})
		if not card.is_empty():
			deck.place_card(slot_index, card.duplicate(true))
		for granted: Dictionary in (entry.get("runes", []) as Array):
			deck.attach_rune(slot_index, RuneEngine.roll_rune(String(granted.get("rune_id", "")), rng))
	for item_id: String in boss_items:
		deck.equip(ItemLibrary.instance(item_id))
	return deck

func _boss_auto_fused_cards() -> Array[Dictionary]:
	var counts: Dictionary = {}
	for id: String in rejected_skills:
		if DealCardLibrary.by_id(id).is_empty():
			continue
		counts[id] = int(counts.get(id, 0)) + 1
	var result: Array[Dictionary] = []
	for id in counts:
		var rank_counts := [0, int(counts[id]), 0, 0, 0, 0]
		for rank in range(1, 5):
			rank_counts[rank + 1] += rank_counts[rank] / 2
			rank_counts[rank] %= 2
		for rank in range(1, 6):
			for _copy in rank_counts[rank]:
				result.append(DealCardLibrary.instance(String(id), rank))
	return result

func _begin_boss_battle() -> void:
	if state != "boss_preview":
		return
	_clear_overlay()
	get_tree().paused = false
	state = "boss"
	current_interactable.clear()
	interaction_text.visible = false
	for enemy: Node in combat.active_enemies.duplicate():
		if is_instance_valid(enemy):
			enemy.queue_free()
	combat.active_enemies.clear()
	combat.enemy_spatial.clear()
	for orb: Node in get_tree().get_nodes_in_group("xp_orbs"):
		if is_instance_valid(orb):
			orb.queue_free()
	var boss_position: Vector2 = _find_visible_boss_position()
	boss = ENEMY_SCRIPT.new()
	var power_level := float(cycle_number + level) * 0.7
	var boss_debts: Array[String] = rejected_skills.duplicate()
	for inherited_skill: Dictionary in boss_advancement_skills:
		boss_debts.append(String(inherited_skill["module"]))
	boss.setup(self, player, "demon_king", 4, power_level, boss_debts, true, false, boss_items)
	boss.position = boss_position
	gameplay_root.add_child(boss)
	boss.use_external_deal_cycle(true)
	boss_cycle = CYCLE_CONTROLLER_SCRIPT.new()
	# v2: 마왕에게도 과열·재진입·빚 RELOAD가 그대로 적용된다. 배율만 ×0.6 (§6.2).
	boss_cycle.setup(self, boss, boss_factory, true, true, run_cycle_seed + 5701)
	gameplay_root.add_child(boss_cycle)
	boss_panel.visible = true
	update_boss_health(boss.health, boss.max_health, boss.shield, boss.max_shield)
	_show_banner("딜싸이클 마왕 · %d칸 / 각인 %d개 / RELOAD ×%.1f" % [boss_factory.slots.size(), boss_factory.total_rune_count(), GameTuning.BOSS_RELOAD_MUL], GamePalette.RED, 4.0)
	spawn_burst(boss_position, GamePalette.RED, 52, 390.0, 1.2)
	play_sound("boss", 0.0)
	shake_camera(12.0, 0.55)

func _find_visible_boss_position() -> Vector2:
	var offsets: Array[Vector2] = [
		Vector2(330.0, 0.0), Vector2(-330.0, 0.0),
		Vector2(0.0, -285.0), Vector2(0.0, 285.0)
	]
	for offset: Vector2 in offsets:
		var candidate := player.global_position + offset
		if world.is_walkable(candidate):
			return candidate
	return world.find_walkable_near(player.global_position, rng, 260.0, 320.0)

func update_boss_health(current: float, maximum: float, shield: float, maximum_shield: float) -> void:
	if not is_instance_valid(boss_fill):
		return
	var health_ratio := clampf(current / maxf(maximum, 1.0), 0.0, 1.0)
	boss_fill.size.x = 560.0 * health_ratio
	var shield_ratio := 0.0 if maximum_shield <= 0.0 else clampf(shield / maximum_shield, 0.0, 1.0)
	boss_shield_fill.size = Vector2(560.0 * shield_ratio, 12.0)
	boss_shield_fill.visible = shield_ratio > 0.0
	var boss_heat := int(boss_cycle.heat) if is_instance_valid(boss_cycle) else 0
	boss_text.text = "딜싸이클 마왕  %d%%  ·  %d칸  ·  과열 %d" % [int(health_ratio * 100.0), GameTuning.BOSS_SLOT_COUNT, boss_heat]

func _on_player_health_changed(current: float, maximum: float) -> void:
	if not is_instance_valid(health_fill):
		return
	var ratio := clampf(current / maxf(maximum, 1.0), 0.0, 1.0)
	health_fill.size.x = (HUD_HEALTH_TRACK.size.x - 4.0) * ratio
	# 게이지 위에 흰 숫자를 얹으므로 채움색은 한 단계 어둡게 씁니다 (⑰ 가독성).
	health_fill.color = GamePalette.GREEN.darkened(0.34) if ratio > 0.52 else GamePalette.YELLOW.darkened(0.3) if ratio > 0.25 else GamePalette.RED.darkened(0.18)
	health_text.text = "%d / %d" % [int(ceil(current)), int(maximum)]

func _on_player_shield_changed(charges: int) -> void:
	if is_instance_valid(shield_text) and is_instance_valid(player):
		shield_text.text = "수호 %d  ·  부활 %d" % [charges, player.rollback_charges]

func _build_result_deal_cycle(parent: Control) -> void:

	# 그대로 재사용합니다. 상호작용만 끈 읽기 전용 레일이므로 스파인 라인·인라인 다리·카드 계층이
	# 공장 화면과 완전히 같은 모습입니다. 레일 끝 RELOAD 마감 패널은 3차 피드백⑬으로 제거했고,
	# 그 덕에 4~5칸까지도 스크롤 없이 화면 안에 가운데 정렬로 들어옵니다.
	var frame := Panel.new()
	frame.position = Vector2(RESULT_BODY_X, RESULT_RAIL_TOP)
	frame.size = Vector2(RESULT_BODY_W, FACTORY_RAIL_ROW_H + 24.0)
	frame.add_theme_stylebox_override("panel", _panel_style(UI_PANEL_BG, UI_EDGE_SOFT, 0, UI_BORDER_FRAME))
	parent.add_child(frame)
	var scroll := ScrollContainer.new()
	scroll.position = Vector2(12.0, 12.0)
	scroll.size = Vector2(RESULT_BODY_W - 24.0, FACTORY_RAIL_ROW_H)
	scroll.horizontal_scroll_mode = ScrollContainer.SCROLL_MODE_AUTO
	scroll.vertical_scroll_mode = ScrollContainer.SCROLL_MODE_DISABLED
	frame.add_child(scroll)
	if factory == null or factory.slots.is_empty():
		var empty_rail := HBoxContainer.new()
		empty_rail.alignment = BoxContainer.ALIGNMENT_CENTER
		scroll.add_child(empty_rail)
		var empty := _label("기록된 딜싸이클이 없습니다.", UI_HEADING_SIZE, GamePalette.MUTED)
		empty.custom_minimum_size = Vector2(scroll.size.x, FACTORY_RAIL_ROW_H)
		empty.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
		empty.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		empty_rail.add_child(empty)
		return
	var slot_count := factory.slots.size()
	var content_width := float(slot_count) * FACTORY_RAIL_CARD_W + float(maxi(0, slot_count - 1)) * FACTORY_RAIL_BRIDGE_W
	# §12 기지 항목(5칸+ 가로 스크롤)은 축소 레일로 시도했다가 되돌렸습니다.
	# Control.scale로 0.75배 하면 6칸이 들어오긴 하지만 픽셀 폰트가 소수 배율에서
	# 뭉개져 지속/RELOAD 숫자를 읽을 수 없습니다(캡처로 확인). 실물 크기를 유지하고
	# 대신 가로 스크롤바가 실제로 보이도록 전역 스크롤바 테마를 고쳤습니다(P2-9).
	var rail := HBoxContainer.new()
	rail.alignment = BoxContainer.ALIGNMENT_CENTER
	rail.add_theme_constant_override("separation", 0)
	scroll.add_child(rail)
	# 칸이 적으면 레일을 가운데로 모으고, 많으면 가로 스크롤로 확인합니다.
	rail.custom_minimum_size = Vector2(maxf(content_width, scroll.size.x), FACTORY_RAIL_ROW_H)
	for slot_index in slot_count:
		_build_factory_rail_slot(rail, slot_index, false)
		if slot_index < slot_count - 1:
			_build_factory_rail_bridge(rail, "built")

	_clear_overlay()
	get_tree().paused = false
