# 카드 / 칸 / 덱 — 부록 C-1의 데이터 모델

# **칸(slot) 객체가 각인 스택을 소유한다.** 덱은 칸 객체의 배열이고, 칸 위치 교환은
# 배열 원소를 통째로 바꾸는 것이므로 각인이 칸과 함께 따라간다(요구사항 그대로).
# 카드만 옮기는 조작(move_card)은 `card` 필드만 교환하므로 각인은 제자리에 남는다.
# 두 조작이 공존한다는 것이 부록 C-1의 두 번째 요구다.
#
# slot := {"card": Dictionary, "runes": Array[Dictionary]}
# rune instance := {"id": String, "p": float, "mag": float}

const DEFAULT_CARD: Dictionary = {
	"id": "basic", "damage": 1.0, "reload": 0.18, "duration": 0.72,
	"element": "", "form": ""
}

static func make_slot(card: Dictionary = {}, runes: Array = []) -> Dictionary:
	var resolved: Dictionary = DEFAULT_CARD.duplicate(true)
	for key in card.keys():
		resolved[key] = card[key]
	var stack: Array = []
	for inst in runes:
		stack.append((inst as Dictionary).duplicate(true))
	return {"card": resolved, "runes": stack}

static func make_deck(slots: Array) -> Array:
	var deck: Array = []
	for entry in slots:
		var slot: Dictionary = entry
		deck.append(make_slot(slot.get("card", {}), slot.get("runes", [])))
	return deck

## 칸 위치 교환 — 각인이 칸과 함께 이동한다 (부록 C-1).
static func swap_slots(deck: Array, a: int, b: int) -> bool:
	if a < 0 or b < 0 or a >= deck.size() or b >= deck.size() or a == b:
		return false
	var tmp: Variant = deck[a]
	deck[a] = deck[b]
	deck[b] = tmp
	return true

## 카드만 교환 — 각인은 원래 칸에 남는다 (부록 C-1의 [카드 이동]).
static func move_card(deck: Array, a: int, b: int) -> bool:
	if a < 0 or b < 0 or a >= deck.size() or b >= deck.size() or a == b:
		return false
	var slot_a: Dictionary = deck[a]
	var slot_b: Dictionary = deck[b]
	var tmp: Variant = slot_a["card"]
	slot_a["card"] = slot_b["card"]
	slot_b["card"] = tmp
	return true

static func same_id_count(slot: Dictionary, id: String) -> int:
	var n := 0
	for inst in (slot.get("runes", []) as Array):
		if String((inst as Dictionary).get("id", "")) == id:
			n += 1
	return n

## 각인 부착. 상한(총 개수·같은 id 개수)을 넘으면 false를 돌려주고 아무것도 안 한다.
static func attach_rune(slot: Dictionary, inst: Dictionary) -> bool:
	var id := String(inst.get("id", ""))
	if not RUNES.has(id):
		return false
	var stack: Array = slot.get("runes", [])
	if stack.size() >= RUNE_STACK_CAP:
		return false
	if same_id_count(slot, id) >= SAME_ID_STACK_CAP:
		return false
	stack.append(inst.duplicate(true))
	slot["runes"] = stack
	return true

## 드래프트: 확률·크기를 범위 안에서 굴려 각인 **인스턴스**를 만든다(§3.3 도입부).
static func roll_rune(id: String, rng: RandomNumberGenerator) -> Dictionary:
	if not RUNES.has(id):
		return {}
	var def: Dictionary = RUNES[id]
	var p: float = float(def["p_min"])
	if float(def["p_max"]) > float(def["p_min"]):
		p = rng.randf_range(float(def["p_min"]), float(def["p_max"]))
	var mag: float = float(def["mag_min"])
	if float(def["mag_max"]) > float(def["mag_min"]):
		mag = rng.randf_range(float(def["mag_min"]), float(def["mag_max"]))
	return {"id": id, "p": p, "mag": mag}

static func all_rune_ids() -> Array[String]:
	var ids: Array[String] = []
	for id: String in RUNES.keys():
		ids.append(id)
	return ids

static func ids_by_rarity(rarity: String) -> Array[String]:
	var ids: Array[String] = []
	for id: String in RUNES.keys():
		if String((RUNES[id] as Dictionary)["rarity"]) == rarity:
			ids.append(id)
	return ids

static func ids_by_family(family: String) -> Array[String]:
	var ids: Array[String] = []
	for id: String in RUNES.keys():
		if String((RUNES[id] as Dictionary)["family"]) == family:
			ids.append(id)
	return ids

# =============================================================================
# 확률 규칙 (§3.4) — 전부 순수 함수. W6 미리보기가 그대로 호출한다.
# =============================================================================

## 같은 id 사본들의 합성 확률. 여집합 곱 + k번째 사본 DUP_P_FALLOFF^(k-1) 감쇠.
static func merged_probability(instances: Array) -> float:
	var complement := 1.0
	for i in instances.size():
		var inst: Dictionary = instances[i]
		var p: float = clampf(float(inst.get("p", 0.0)) * pow(DUP_P_FALLOFF, i), 0.0, 1.0)
		complement *= (1.0 - p)
	return 1.0 - complement

## 같은 id 사본들의 합성 크기. k번째 사본은 DUP_MAG_FALLOFF^(k-1)만 기여한다.
static func merged_magnitude(instances: Array) -> float:
	var total := 0.0
	for i in instances.size():
		var inst: Dictionary = instances[i]
		total += float(inst.get("mag", 0.0)) * pow(DUP_MAG_FALLOFF, i)
	return total

## 과밀 페널티. 한 칸에 RUNE_SLOTS_PER_SLOT을 넘겨 담으면 그 칸 전체 확률이 깎인다.
static func congestion_scale(rune_count: int) -> float:
	return pow(CONGESTION_FALLOFF, float(maxi(0, rune_count - RUNE_SLOTS_PER_SLOT)))

## 최종 발동 확률. 순서: 합성 → 과밀 → 과열 감쇠 → P_CAP 클램프(§3.4).
static func effective_probability(base_p: float, deviation_load: float, congestion: float) -> float:
	return clampf(base_p * congestion * pow(HEAT_DECAY, deviation_load), 0.0, P_CAP)

static func heat_from_load(deviation_load: float, heat_bonus: int = 0) -> int:
	return clampi(int(floor(deviation_load)) + heat_bonus, 0, HEAT_MAX)

static func damage_multiplier(heat: int, reentry: int) -> float:
	return (1.0 + float(heat) * HEAT_DAMAGE) * pow(REENTRY_FALLOFF, float(reentry))

## 조건 각인의 조건 판정. false면 굴리지도 않는다 → 과열 예산 소모 없음(§3.4).
static func condition_ok(cond: String, ctx: Dictionary) -> bool:
	match cond:
		"always":
			return true
		"first":
			return int(ctx.get("reentry", 0)) == 0
		"reentry":
			return int(ctx.get("reentry", 0)) > 0
		"kill":
			return bool(ctx.get("killed", false))
		"heat_gate":
			return int(ctx.get("heat", 0)) >= HEAT_GATE_MIN
		"prev_slot":
			return int(ctx.get("prev_index", -1)) >= 0
		"prev_same_element":
			var prev := String(ctx.get("prev_element", ""))
			var here := String(ctx.get("element", ""))
			return prev != "" and prev == here
		_:
			return true

static func slot_element(slot: Dictionary) -> String:
	return String((slot.get("card", {}) as Dictionary).get("element", ""))

static func slot_form(slot: Dictionary) -> String:
	return String((slot.get("card", {}) as Dictionary).get("form", ""))

## 인접 공명: 이웃한 칸이 같은 원소면 이 칸의 피해 +15%.
static func resonance_bonus(deck: Array, index: int) -> float:
	var here := slot_element(deck[index])
	if here == "":
		return 0.0
	if index > 0 and slot_element(deck[index - 1]) == here:
		return RESONANCE_DAMAGE
	if index < deck.size() - 1 and slot_element(deck[index + 1]) == here:
		return RESONANCE_DAMAGE
	return 0.0

## 결속 구간: 같은 원소가 BOND_MIN_RUN칸 이상 연속인 구간. 그 안의 각인은 과열이 싸다.
static func bond_mask(deck: Array) -> Array[bool]:
	var mask: Array[bool] = []
	for i in deck.size():
		mask.append(false)
	var run_start := 0
	while run_start < deck.size():
		var element := slot_element(deck[run_start])
		var run_end := run_start
		while run_end + 1 < deck.size() and slot_element(deck[run_end + 1]) == element:
			run_end += 1
		if element != "" and (run_end - run_start + 1) >= BOND_MIN_RUN:
			for i in range(run_start, run_end + 1):
				mask[i] = true
		run_start = run_end + 1
	return mask

## 삼각 배열: 1·3·5칸이 같은 형태 태그면 전체 RELOAD −12%.
static func triangle_ok(deck: Array) -> bool:
	if deck.size() < 5:
		return false
	var f := slot_form(deck[0])
	return f != "" and f == slot_form(deck[2]) and f == slot_form(deck[4])

## 원소 반응 7쌍(§4.5). 직전 실행 칸의 원소 → 이번 칸의 원소.
## 직전 칸이 비어 있으면(빈 칸 = 기본 베기 = 태그 없음) 어떤 반응도 서지 않는다.

# =============================================================================
# 각인 판정 — 순수 함수 (W1 완료 기준 ②)
# =============================================================================
# resolve(slot_runes, ctx, rng) -> {repeat, delta, reverse, fired[], ...}
#
# ctx 필수 키: deviation_load(float), heat(int), reentry(int), bond(bool),
#              element(String), prev_element(String), prev_index(int),
#              killed(bool), slot_index(int), bookmark(int)
#
# 굴림 순서 = slot_runes의 부착 순서. 한 칸 안에서도 각인이 하나 터질 때마다
# deviation_load가 즉시 올라 뒤에 굴리는 각인의 확률이 그 자리에서 깎인다(§3.2 6번).
static func resolve(slot_runes: Array, ctx: Dictionary, rng: RandomNumberGenerator) -> Dictionary:
	var out: Dictionary = {
		"repeat": 0, "delta": 0, "reverse": false, "jump_to": -1, "set_bookmark": -1,
		"fired": [], "fire_cost": 0.0, "fire_count": 0,
		"damage_bonus": 0.0, "damage_mult": 1.0, "range_bonus": 0.0,
		"pierce_bonus": 0, "duration_mult": 1.0, "debt_delta": 0.0,
		"free_reload": false, "link_next": false,
		"echo_power": 0.0, "chorus_power": 0.0, "overlap_power": 0.0,
		"afterburn": false, "toll": false
	}
	if slot_runes.is_empty():
		return out

	var groups := _group_runes(slot_runes)
	var congestion := congestion_scale(slot_runes.size())
	var load := float(ctx.get("deviation_load", 0.0))
	var bond := bool(ctx.get("bond", false))
	var fire_cost := (BOND_FIRE_COST if bond else 1.0)

	for group: Dictionary in groups:
		var id := String(group["id"])
		var def: Dictionary = RUNES[id]
		var instances: Array = group["instances"]

		if not condition_ok(String(def["cond"]), ctx):
			continue

		var magnitude := merged_magnitude(instances)

		# 확정(패시브) 각인: 굴리지 않고 항상 적용. 과열 비용 0.
		if not bool(def["roll"]):
			_apply_rune(out, id, magnitude, instances.size(), ctx)
			continue

		var base_p := merged_probability(instances)
		var p := effective_probability(base_p, load, congestion)
		if rng.randf() >= p:
			continue

		load += fire_cost
		out["fire_cost"] = float(out["fire_cost"]) + fire_cost
		out["fire_count"] = int(out["fire_count"]) + 1
		(out["fired"] as Array).append(id)
		_apply_rune(out, id, magnitude, instances.size(), ctx)

	return out

## 부착 순서를 유지하면서 같은 id를 묶는다. Dictionary가 삽입 순서를 보존하므로 결정적.
static func _group_runes(slot_runes: Array) -> Array:
	var order: Array[String] = []
	var buckets: Dictionary = {}
	for entry in slot_runes:
		var inst: Dictionary = entry
		var id := String(inst.get("id", ""))
		if not RUNES.has(id):
			continue
		if not buckets.has(id):
			buckets[id] = []
			order.append(id)
		var bucket: Array = buckets[id]
		if bucket.size() >= SAME_ID_STACK_CAP:
			continue
		bucket.append(inst)
	var groups: Array = []
	for id in order:
		groups.append({"id": id, "instances": buckets[id]})
	return groups

static func _apply_rune(out: Dictionary, id: String, magnitude: float, copies: int, ctx: Dictionary) -> void:
	match id:
		# --- 흐름 ---
		"rewind_1", "rewind_2", "skip_1":
			# 중복해도 delta는 커지지 않는다. 확률만 오른다(부록 C-1 감쇠 설계).
			out["delta"] = int(out["delta"]) + int(FLOW_DELTA[id])
		"repeat", "kill_repeat":
			out["repeat"] = int(out["repeat"]) + 1
		"echo":
			out["echo_power"] = float(out["echo_power"]) + magnitude
		"reverse":
			# 홀수 번 뒤집혀야 실제로 방향이 바뀐다. 같은 칸 2사본이 한 번에 터져도
			# 합성 굴림 1회라 여기서 toggle은 최대 1회다.
			out["reverse"] = not bool(out["reverse"])
		"bookmark":
			var mark := int(ctx.get("bookmark", -1))
			var here := int(ctx.get("slot_index", 0))
			if mark >= 0 and mark != here:
				out["jump_to"] = mark
			else:
				out["set_bookmark"] = here
		# --- 동시 ---
		"link_next":
			out["link_next"] = true
		"chorus":
			out["chorus_power"] = float(out["chorus_power"]) + magnitude
		"overlap":
			out["overlap_power"] = float(out["overlap_power"]) + magnitude
		# --- 조건 ---
		"tag_chain", "first_strike":
			out["damage_bonus"] = float(out["damage_bonus"]) + magnitude
		"heat_gate":
			# 곱연산. 중복 사본은 크기 감쇠분만큼만 더 곱한다.
			out["damage_mult"] = float(out["damage_mult"]) * (1.0 + (magnitude - 1.0))
		"last_call":
			out["debt_delta"] = float(out["debt_delta"]) - magnitude
		"odd_even":
			if int(ctx.get("reentry", 0)) % 2 == 0:   # 1·3·5번째 실행 = 홀수 번째
				out["damage_bonus"] = float(out["damage_bonus"]) + magnitude
			else:
				out["range_bonus"] = float(out["range_bonus"]) + ODD_RANGE_BONUS
		# --- 템포 ---
		"haste":
			out["duration_mult"] = float(out["duration_mult"]) * maxf(0.4, 1.0 - magnitude)
		"refund":
			out["debt_delta"] = float(out["debt_delta"]) - magnitude
		"toll":
			out["toll"] = true
		"free_reload":
			out["free_reload"] = true
		"afterburn":
			out["afterburn"] = true
		# --- 전투 ---
		"edge":
			out["damage_bonus"] = float(out["damage_bonus"]) + magnitude
		"reach":
			out["range_bonus"] = float(out["range_bonus"]) + magnitude
		"barb":
			out["pierce_bonus"] = int(out["pierce_bonus"]) + copies

## 칸에 붙은 확정(패시브) 각인 중 특정 id가 있으면 합성 크기를 돌려준다. 없으면 0.
static func passive_magnitude(slot: Dictionary, id: String) -> float:
	var instances: Array = []
	for entry in (slot.get("runes", []) as Array):
		var inst: Dictionary = entry
		if String(inst.get("id", "")) == id and instances.size() < SAME_ID_STACK_CAP:
			instances.append(inst)
	if instances.is_empty():
		return 0.0
	return merged_magnitude(instances)

# =============================================================================
# 한 사이클 시뮬레이션 (W1 완료 기준 ④ · W2 런타임 · W6 미리보기 공용)
# =============================================================================
# simulate_cycle(deck, seed, opts) -> 실행 궤적 + 총피해 배율 + 과열 곡선 + RELOAD
#
# opts (전부 선택)
#   start_load   : float  이전 사이클의 잔열(afterburn)이 넘겨준 시작 과열. 기본 0
#   kill_chance  : float  도륙(kill_repeat) 조건 모델링용. 0이면 처치 없음. 기본 0
#   kills        : Array[bool]  결정적으로 처치 여부를 주고 싶을 때(W2가 실제 값 주입)
#   reload_scale : float  보스전 등 외부 배율. 기본 1.0
#   direction    : int    시작 방향. 기본 +1
#
# 반환값은 전부 순수 데이터다. W2는 steps[]를 프레임에 펼치고, W6는 같은 steps[]를
# 레일 위 화살표로 그린다. 두 화면이 어긋나지 않는 유일한 방법이 이 공유다.
static func simulate_cycle(deck: Array, cycle_seed: int, opts: Dictionary = {}) -> Dictionary:
	var n := deck.size()
	var result: Dictionary = {
		"steps": [], "step_count": 0, "damage_total": 0.0, "damage_mul_sum": 0.0,
		"heat_curve": [], "peak_heat": 0, "end_heat": 0, "deviation_load": 0.0,
		"fired": [], "fire_count": 0, "reload_debt": 0.0, "reload": 0.0,
		"overloaded": false, "end_reason": "empty", "carry_heat": 0,
		"visited": [], "reactions": []
	}
	if n <= 0:
		return result

	var rng := RandomNumberGenerator.new()
	rng.seed = cycle_seed

	var bonds := bond_mask(deck)
	var triangle := triangle_ok(deck)

	var cursor := 0
	var direction := int(opts.get("direction", 1))
	if direction == 0:
		direction = 1
	var load := float(opts.get("start_load", 0.0))
	var heat_bonus := 0
	var debt := 0.0
	var step_count := 0
	var repeats_left := 0
	var carried_delta := 0
	var bookmark := -1
	var prev_index := -1
	var afterburn_seen := false
	var pending_end := false
	var kill_chance := float(opts.get("kill_chance", 0.0))
	var kills: Array = opts.get("kills", [])
	var guard := 0

	var executed: Array[int] = []
	for i in n:
		executed.append(0)

	while true:
		guard += 1
		if guard > HARD_LOOP_GUARD:
			result["end_reason"] = "guard"
			break

		# ---------------------------------------------------------- 스텝 실행
		var reentry := executed[cursor]
		executed[cursor] += 1
		step_count += 1

		var slot: Dictionary = deck[cursor]
		var card: Dictionary = slot.get("card", DEFAULT_CARD)
		var element := slot_element(slot)
		var prev_element := (slot_element(deck[prev_index]) if prev_index >= 0 else "")

		var reaction := reaction_of(prev_element, element)
		if reaction == "overcharge":
			heat_bonus += OVERCHARGE_HEAT_BONUS

		# §3.2는 카드 실행(4) 뒤에 각인을 굴리지만(6), 조건·전투 각인은 **이 칸의**
		# 피해를 바꾸는 물건이라 판정이 카드보다 앞서야 수치가 성립한다. 그래서
		# 순서를 "칸 진입 → 각인 판정 → 카드 실행(버프 반영) → 이동"으로 바꿨다.
		# 피해에 쓰는 heat는 여전히 **굴리기 전** 값이라 §3.2 3번의 의미는 보존된다.
		var heat := heat_from_load(load, heat_bonus)

		var killed := false
		if kills.size() > step_count - 1:
			killed = bool(kills[step_count - 1])
		elif kill_chance > 0.0:
			killed = rng.randf() < kill_chance

		var ctx: Dictionary = {
			"deviation_load": load, "heat": heat, "reentry": reentry,
			"bond": bonds[cursor], "element": element, "prev_element": prev_element,
			"prev_index": prev_index, "killed": killed, "slot_index": cursor,
			"bookmark": bookmark
		}
		var r := resolve(slot.get("runes", []), ctx, rng)

		load += float(r["fire_cost"])
		if bool(r["afterburn"]):
			afterburn_seen = true
		if int(r["set_bookmark"]) >= 0:
			bookmark = int(r["set_bookmark"])

		# ------------------------------------------------------------ 피해 계산
		var resonance := resonance_bonus(deck, cursor)
		var damage_mul := damage_multiplier(heat, reentry)
		damage_mul *= (1.0 + resonance)
		damage_mul *= (1.0 + float(r["damage_bonus"]))
		damage_mul *= float(r["damage_mult"])

		var base_damage := float(card.get("damage", 1.0))
		var step_damage := base_damage * damage_mul

		# 동시 발동분 (§3.3 B)
		var link_target := -1
		if bool(r["link_next"]):
			var candidate := cursor + direction
			if candidate >= 0 and candidate < n:
				link_target = candidate
		if link_target >= 0:
			var link_card: Dictionary = (deck[link_target] as Dictionary).get("card", DEFAULT_CARD)
			step_damage += float(link_card.get("damage", 1.0)) * damage_mul * LINK_POWER
			executed[link_target] += 1
			debt += float(link_card.get("reload", 0.0))
		if float(r["echo_power"]) > 0.0 and prev_index >= 0:
			var echo_card: Dictionary = (deck[prev_index] as Dictionary).get("card", DEFAULT_CARD)
			step_damage += float(echo_card.get("damage", 1.0)) * damage_mul * float(r["echo_power"])
		if float(r["overlap_power"]) > 0.0 and prev_index >= 0:
			var over_card: Dictionary = (deck[prev_index] as Dictionary).get("card", DEFAULT_CARD)
			step_damage += float(over_card.get("damage", 1.0)) * damage_mul * float(r["overlap_power"])
		if float(r["chorus_power"]) > 0.0:
			var chorus_sum := 0.0
			if cursor > 0:
				chorus_sum += float((deck[cursor - 1] as Dictionary).get("card", DEFAULT_CARD).get("damage", 1.0))
			if cursor < n - 1:
				chorus_sum += float((deck[cursor + 1] as Dictionary).get("card", DEFAULT_CARD).get("damage", 1.0))
			step_damage += chorus_sum * damage_mul * float(r["chorus_power"])
		if reaction == "shock":
			step_damage += base_damage * damage_mul * SHOCK_SPLASH

		# ------------------------------------------------------------ 빚 누적
		if not bool(r["free_reload"]):
			debt += float(card.get("reload", 0.0))
		debt += float(r["debt_delta"])
		# *→초 공명 흡수(§4.5): 초능력 칸이 앞 칸의 잔향을 흡수해 빚을 깎는다.
		# 빚 모델에 얹는 이유 — RELOAD는 `debt × (1 + 과열)`이라 최종값에서 빼면
		# 과열이 높을수록 감면이 커지는(설계에 없는) 보너스가 생긴다.
		if reaction == "resonant_drain":
			debt -= RESONANT_DRAIN_RELOAD
		debt = maxf(0.0, debt)

		var range_bonus := float(r["range_bonus"])
		if reaction == "steam":
			range_bonus += STEAM_RANGE_BONUS

		# 상태 위력(potency) — v3 신규 채널. combat_resolver가 상태를 부여할 때 곱한다.
		# **여기서 피해를 만들지 않는다.** 편집 화면 96표본 미리보기가 이 값을 그대로 읽는다.
		var potency := reaction_potency(reaction)
		# 경직 가산 — 빙→타 쇄빙 준비. 카드 자체의 경직(knockback_profile)에 더해진다.
		var stun_bonus := (SHATTER_PREP_STUN if reaction == "shatter_prep" else 0.0)

		# ------------------------------------------------------------ 궤적 기록
		var step_record: Dictionary = {
			"index": step_count - 1, "slot": cursor, "card_id": String(card.get("id", "")),
			"reentry": reentry, "heat": heat, "damage_mul": damage_mul,
			"damage": step_damage, "direction": direction,
			"duration": float(card.get("duration", 0.0)) * float(r["duration_mult"]),
			"fired": (r["fired"] as Array).duplicate(),
			"delta": int(r["delta"]), "repeat": int(r["repeat"]),
			"reverse": bool(r["reverse"]), "link": link_target,
			"range_bonus": range_bonus, "pierce_bonus": int(r["pierce_bonus"]),
			"reaction": reaction, "bond": bonds[cursor], "debt": debt,
			"element": element, "prev_element": prev_element,
			"potency": potency, "stun_bonus": stun_bonus
		}
		(result["steps"] as Array).append(step_record)
		(result["heat_curve"] as Array).append(heat)
		(result["visited"] as Array).append(cursor)
		if reaction != "":
			(result["reactions"] as Array).append(reaction)
		for id in (r["fired"] as Array):
			(result["fired"] as Array).append(id)
		result["damage_total"] = float(result["damage_total"]) + step_damage
		result["damage_mul_sum"] = float(result["damage_mul_sum"]) + damage_mul
		result["fire_count"] = int(result["fire_count"]) + int(r["fire_count"])
		result["peak_heat"] = maxi(int(result["peak_heat"]), heat)

		prev_index = cursor

		# 역행이 1칸에 도달해 실행까지 마쳤으면 여기서 사이클 종료(§3.2 경계 표).
		if pending_end:
			result["end_reason"] = "reverse_home"
			break

		# 하드 안전망 3층: STEP_CAP (§3.9)
		if step_count >= STEP_CAP:
			result["end_reason"] = "overload"
			result["overloaded"] = true
			break

		# ---------------------------------------------------------- 이동 결정
		# 우선순위: repeat → delta 합 → reverse (§3.4 "순서")
		if int(r["repeat"]) > 0 and repeats_left == 0:
			repeats_left = mini(int(r["repeat"]), REPEAT_CAP)
		if repeats_left > 0:
			repeats_left -= 1
			carried_delta += int(r["delta"])   # 되감기는 그 스텝의 delta로 이월(§3.2 표)
			if bool(r["reverse"]):
				direction = -direction
			continue

		if bool(r["reverse"]):
			direction = -direction

		var next_index := cursor
		if int(r["jump_to"]) >= 0:
			next_index = int(r["jump_to"])
			bookmark = -1
			carried_delta = 0
		else:
			var move := 1 + int(r["delta"]) + carried_delta
			if link_target >= 0:
				move += 1              # 연결한 칸은 이미 실행됐으므로 통과한다
			carried_delta = 0
			next_index = cursor + direction * move

			if direction > 0:
				if next_index > n - 1:
					result["end_reason"] = "complete"
					break
				if next_index < 0:
					next_index = 0     # 1칸 되감기는 감싸지 않고 클램프(§3.2 표)
			else:
				if next_index <= 0:
					next_index = 0
					pending_end = true # 역행이 1칸에 닿으면 그 칸 실행 후 종료
				elif next_index > n - 1:
					next_index = n - 1

		cursor = clampi(next_index, 0, n - 1)

	# ------------------------------------------------------------- 사이클 종료
	var end_heat := heat_from_load(load, heat_bonus)
	var reload := debt * (1.0 + float(end_heat) * HEAT_RELOAD)
	if triangle:
		reload *= (1.0 - TRIANGLE_RELOAD_DISCOUNT)
	reload *= float(opts.get("reload_scale", 1.0))
	if bool(result["overloaded"]):
		reload = RELOAD_CAP * float(opts.get("reload_scale", 1.0))
	reload = clampf(reload, 0.0, RELOAD_CAP)

	result["step_count"] = step_count
	result["deviation_load"] = load
	result["end_heat"] = end_heat
	result["reload_debt"] = debt
	result["reload"] = reload
	result["carry_heat"] = (int(floor(float(end_heat) / 2.0)) if afterburn_seen else 0)
	if String(result["end_reason"]) == "empty":
		result["end_reason"] = "complete"
	return result

## 궤적 지문. 시드 결정성 검증과 W6 미리보기 캐시 키에 쓴다.
static func trace_signature(cycle: Dictionary) -> String:
	var parts: Array[String] = []
	for entry in (cycle.get("steps", []) as Array):
		var step: Dictionary = entry
		parts.append("%d:%d:%d:%.4f:%s" % [
			int(step["slot"]), int(step["reentry"]), int(step["heat"]),
			float(step["damage_mul"]), ",".join(step["fired"] as Array)
		])
	parts.append("R%.4f" % float(cycle.get("reload", 0.0)))
	return "|".join(parts)

## 편집 화면 요약. W6가 칸을 옮길 때마다 이걸 다시 부르면 된다.
static func preview(deck: Array, cycle_seed: int, samples: int = 64, opts: Dictionary = {}) -> Dictionary:
	var steps_sum := 0.0
	var damage_sum := 0.0
	var reload_sum := 0.0
	var heat_sum := 0.0
	var overload := 0
	var count := maxi(1, samples)
	for i in count:
		var cycle := simulate_cycle(deck, cycle_seed + i, opts)
		steps_sum += float(cycle["step_count"])
		damage_sum += float(cycle["damage_total"])
		reload_sum += float(cycle["reload"])
		heat_sum += float(cycle["peak_heat"])
		if bool(cycle["overloaded"]):
			overload += 1
	return {
		"samples": count,
		"mean_steps": steps_sum / float(count),
		"mean_damage": damage_sum / float(count),
		"mean_reload": reload_sum / float(count),
		"mean_peak_heat": heat_sum / float(count),
		"overload_rate": float(overload) / float(count),
		"bond": bond_mask(deck),
		"triangle": triangle_ok(deck)
	}
