class_name WorldGrid
extends Node2D

const WFC_SCRIPT := preload("res://scripts/wfc_chunk_generator.gd")
const TERRAIN_ATLAS := preload("res://art/generated/world/terrain-atlas-wfc-v2-runtime.png")

const TILE := 40
const FEATURE_CHUNK := 800
const VIEW_HALF := Vector2(760.0, 460.0)
# 런타임 아틀라스는 셀마다 1픽셀(=원본 2픽셀) 어두운 테두리를 갖고 있습니다.
# 셀 전체를 그대로 그리면 그 테두리가 타일마다 겹쳐 화면 전체에 40px 격자
# 그물이 보입니다(3차 피드백 "맵이 깨져 보인다"의 절반). 셀 안쪽으로 그만큼
# 잘라내면 잔디가 하나의 면으로 이어집니다.
const ATLAS_CELL_INSET := 2
const ATLAS_COLUMNS := 5
const ATLAS_ROWS := 4
# 시작 성 동쪽 고정 물길. 다리는 한 줄(가로 3칸)만 놓습니다. 아틀라스의 다리
# 그림은 데크가 좌우로 지나가고 위아래가 물이므로, 여러 줄을 겹치면 물줄기가
# 낀 판자 여러 장으로 보입니다(개편 전 모습).
const RIVER_LEFT := 19
const RIVER_RIGHT := 21
const RIVER_TOP := -9
const RIVER_BOTTOM := 9
const RIVER_BRIDGE_ROW := 0
const SAFE_ZONE_DEMON := 210.0
const SAFE_ZONE_CASTLE := 190.0
const SAFE_ZONE_CAMP := 175.0
const STARTER_CASTLE_POSITION := Vector2(250.0, -250.0)
# 지형은 논문의 Simple Tiled WFC를 스트리밍 청크 방식으로 확장해 끝없이 생성됩니다.
# 주요 목적지는 시작 지점에서 충분히 멀고, 정확한 축 위가 아닌 각 방위권 어딘가에
# 놓여 탐험으로 찾아야 합니다.
const DEMON_CASTLE_POSITION := Vector2(6840.0, -5260.0)
const TRIAL_CAMPS := {
	"north": Vector2(520.0, -4180.0),
	"east": Vector2(4320.0, 860.0),
	"south": Vector2(-640.0, 4460.0),
	"west": Vector2(-4250.0, -780.0)
}

var player: Node2D
var night_amount := 0.0
var opened_features: Dictionary = {}
var cleared_trial_camps: Dictionary = {}
var last_draw_center := Vector2(1000000.0, 1000000.0)
var last_draw_night := -1.0
var terrain_generator = WFC_SCRIPT.new()
var tile_sources: Array[Rect2] = []
var tile_turns: PackedInt32Array = PackedInt32Array()

func _init() -> void:
	# 목적지 바닥은 안전 타일로 덮이므로 그 자리에는 호수를 만들지 않습니다.
	# 이렇게 하지 않으면 덮개가 호수를 잘라 물 한가운데 사각형 잔디가 생깁니다.
	terrain_generator.add_dry_zone(DEMON_CASTLE_POSITION / float(TILE), SAFE_ZONE_DEMON / float(TILE))
	terrain_generator.add_dry_zone(STARTER_CASTLE_POSITION / float(TILE), SAFE_ZONE_CASTLE / float(TILE))
	for camp_position: Vector2 in TRIAL_CAMPS.values():
		terrain_generator.add_dry_zone(camp_position / float(TILE), SAFE_ZONE_CAMP / float(TILE))
	# 손으로 얹은 시작 물길 자리에는 WFC 호수를 만들지 않습니다. 겹치면 덮개
	# 바깥에서 호수 물가와 물길 물가가 서로 다른 모양으로 맞닿습니다.
	terrain_generator.add_dry_rect(Rect2(
		float(RIVER_LEFT) - 2.0, float(RIVER_TOP) - 2.0,
		float(RIVER_RIGHT - RIVER_LEFT) + 5.0, float(RIVER_BOTTOM - RIVER_TOP) + 5.0
	))
	_build_tile_atlas_table()

func setup(player_node: Node2D) -> void:
	player = player_node

func _ready() -> void:
	z_index = -20
	texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
	queue_redraw()

func _build_tile_atlas_table() -> void:
	tile_sources.clear()
	tile_turns.resize(0)
	var inset := float(ATLAS_CELL_INSET)
	for tile_id in int(terrain_generator.TILE_COUNT):
		var cell: int = int(terrain_generator.tile_atlas_cell(tile_id))
		var column: int = cell % ATLAS_COLUMNS
		var row: int = cell / ATLAS_COLUMNS
		# 아틀라스 크기가 격자 수로 나누어지지 않아도 픽셀이 번지지 않도록
		# 각 셀의 시작·끝을 정수 좌표로 계산한 뒤 안쪽으로 잘라냅니다.
		var left := floori(float(TERRAIN_ATLAS.get_width() * column) / float(ATLAS_COLUMNS))
		var right := floori(float(TERRAIN_ATLAS.get_width() * (column + 1)) / float(ATLAS_COLUMNS))
		var top := floori(float(TERRAIN_ATLAS.get_height() * row) / float(ATLAS_ROWS))
		var bottom := floori(float(TERRAIN_ATLAS.get_height() * (row + 1)) / float(ATLAS_ROWS))
		var usable := minf(inset, minf(float(right - left), float(bottom - top)) * 0.25)
		tile_sources.append(Rect2(
			float(left) + usable, float(top) + usable,
			float(right - left) - usable * 2.0, float(bottom - top) - usable * 2.0
		))
		tile_turns.append(int(terrain_generator.tile_turn(tile_id)))

func _process(_delta: float) -> void:
	if is_instance_valid(player) and player.global_position.distance_squared_to(last_draw_center) >= 32.0 * 32.0:
		last_draw_center = player.global_position
		queue_redraw()

func set_night_amount(value: float) -> void:
	night_amount = clampf(value, 0.0, 1.0)
	var reached_endpoint := (is_zero_approx(night_amount) or is_equal_approx(night_amount, 1.0)) and not is_equal_approx(last_draw_night, night_amount)
	if absf(night_amount - last_draw_night) >= 0.025 or reached_endpoint:
		last_draw_night = night_amount
		queue_redraw()

func set_opened_features(features: Dictionary) -> void:
	opened_features = features

func set_cleared_trial_camps(cleared: Dictionary) -> void:
	cleared_trial_camps = cleared.duplicate()
	queue_redraw()

func get_trial_camps() -> Dictionary:
	return TRIAL_CAMPS.duplicate()

func is_walkable(point: Vector2) -> bool:
	var tile_id := _resolved_tile_id(_world_to_tile(point))
	return not terrain_generator.is_water_tile(tile_id) or terrain_generator.is_bridge_tile(tile_id)

func get_generation_stats() -> Dictionary:
	return terrain_generator.stats()

func get_tile_id(point: Vector2) -> int:
	return _resolved_tile_id(_world_to_tile(point))

func validate_generation_near(point: Vector2) -> Dictionary:
	var chunk: Vector2i = terrain_generator.chunk_for_tile(_world_to_tile(point))
	var local_ok: bool = terrain_generator.validate_chunk(chunk)
	var seams_ok := true
	for offset in [Vector2i.RIGHT, Vector2i.DOWN, Vector2i.LEFT, Vector2i.UP]:
		seams_ok = seams_ok and terrain_generator.validate_neighbor_seam(chunk, chunk + offset)
	return {"chunk":chunk, "local_rules":local_ok, "seams":seams_ok}

func get_nearest_interactable(point: Vector2, radius: float = 92.0) -> Dictionary:
	var nearest: Dictionary = {}
	var nearest_distance := radius
	for feature: Dictionary in _features_near(point, 2):
		if feature["type"] == "chest" and opened_features.has(feature["id"]):
			continue
		var distance := point.distance_to(feature["position"])
		if distance < nearest_distance:
			nearest_distance = distance
			nearest = feature
	return nearest

func get_castle_services(feature_id: String, awakening_boost: bool = false) -> Array[String]:
	var all_services: Array[String] = ["card_shop", "factory_mage", "merchant", "boss_remove", "inn", "awakening_master"]
	var seed_value: int = absi(feature_id.hash())
	var result: Array[String] = ["card_fusion"]
	for step in 3:
		var index: int = (seed_value + step * 7) % all_services.size()
		while result.has(all_services[index]):
			index = (index + 1) % all_services.size()
		result.append(all_services[index])
	# 네 구슬을 모은 뒤에는 대부분의 성에서 전승 현자를 만날 수 있습니다.
	if awakening_boost and not result.has("awakening_master"):
		var chance_roll := (seed_value / 13) % 100
		if chance_roll < 90:
			result[result.size() - 1] = "awakening_master"
	return result

func get_demon_castle_position() -> Vector2:
	return DEMON_CASTLE_POSITION

func find_walkable_near(point: Vector2, rng: RandomNumberGenerator, minimum: float, maximum: float) -> Vector2:
	for _attempt in 24:
		var candidate := point + Vector2.from_angle(rng.randf_range(0.0, TAU)) * rng.randf_range(minimum, maximum)
		if is_walkable(candidate) and candidate.distance_to(DEMON_CASTLE_POSITION) > 240.0:
			return candidate
	return point + Vector2(maximum, 0.0)

func _draw() -> void:
	var center := Vector2.ZERO
	if is_instance_valid(player):
		center = player.global_position
	var start_x := int(floor((center.x - VIEW_HALF.x) / TILE)) * TILE
	var end_x := int(ceil((center.x + VIEW_HALF.x) / TILE)) * TILE
	var start_y := int(floor((center.y - VIEW_HALF.y) / TILE)) * TILE
	var end_y := int(ceil((center.y + VIEW_HALF.y) / TILE)) * TILE

	var y := start_y
	while y <= end_y:
		var x := start_x
		while x <= end_x:
			_draw_tile(Vector2(x, y), Vector2i(floori(float(x) / TILE), floori(float(y) / TILE)))
			x += TILE
		y += TILE
	var center_tile := _world_to_tile(center)
	terrain_generator.trim_cache(terrain_generator.chunk_for_tile(center_tile))

	for feature: Dictionary in _features_near(center, 2):
		if feature["position"].distance_to(center) > 1050.0:
			continue
		match feature["type"]:
			"castle": _draw_castle(feature["position"], false)
			"demon_castle": _draw_castle(feature["position"], true)
			"chest":
				if not opened_features.has(feature["id"]):
					_draw_chest(feature["position"])
			"grove": _draw_grove(feature["position"], int(feature["variant"]))
			"ruin": _draw_ruin(feature["position"])
	for direction: String in TRIAL_CAMPS:
		var camp_position: Vector2 = TRIAL_CAMPS[direction]
		if camp_position.distance_to(center) <= 1050.0:
			_draw_trial_camp(camp_position, direction, bool(cleared_trial_camps.get(direction, false)))

func _draw_tile(tile_position: Vector2, tile_coord: Vector2i) -> void:
	var tile_id := _resolved_tile_id(tile_coord)
	if tile_id < 0 or tile_id >= tile_sources.size():
		tile_id = 0
	var source: Rect2 = tile_sources[tile_id]
	var night_tint := Color.WHITE.lerp(Color("7181a4"), night_amount * 0.48)
	var turn := int(tile_turns[tile_id])
	if turn == 0:
		draw_texture_rect_region(TERRAIN_ATLAS, Rect2(tile_position, Vector2(TILE, TILE)), source, night_tint)
		return
	# 아틀라스에 호수 남동/남서 모서리 그림만 있으므로 북서/북동 모서리는 같은
	# 셀을 90도 단위로 돌려 만듭니다. 정확히 0/±1 성분만 쓰는 행렬이라
	# nearest 샘플링에서 픽셀이 밀리지 않습니다.
	var center := tile_position + Vector2(TILE, TILE) * 0.5
	var basis_x := Vector2(1.0, 0.0)
	var basis_y := Vector2(0.0, 1.0)
	match turn:
		1:
			basis_x = Vector2(0.0, 1.0)
			basis_y = Vector2(-1.0, 0.0)
		2:
			basis_x = Vector2(-1.0, 0.0)
			basis_y = Vector2(0.0, -1.0)
		3:
			basis_x = Vector2(0.0, -1.0)
			basis_y = Vector2(1.0, 0.0)
	draw_set_transform_matrix(Transform2D(basis_x, basis_y, center))
	draw_texture_rect_region(TERRAIN_ATLAS, Rect2(Vector2(TILE, TILE) * -0.5, Vector2(TILE, TILE)), source, night_tint)
	draw_set_transform_matrix(Transform2D.IDENTITY)

func _world_to_tile(point: Vector2) -> Vector2i:
	return Vector2i(floori(point.x / float(TILE)), floori(point.y / float(TILE)))

func _resolved_tile_id(tile_coord: Vector2i) -> int:
	var center := Vector2(tile_coord.x * TILE + TILE * 0.5, tile_coord.y * TILE + TILE * 0.5)
	var starter_landmark := _starter_river_tile(tile_coord)
	if starter_landmark >= 0:
		return starter_landmark
	# 전투 목적지는 WFC 위에 얹는 설계 고정점입니다. 목적지 바닥만 안전한
	# 타일로 덮어 보스·성 입구가 호수 속에 생기는 상황을 방지합니다.
	# 같은 반지름이 호수 레이어의 dry zone으로도 등록돼 있어 덮개가 호수를
	# 잘라내는 일은 생기지 않습니다.
	if center.distance_to(DEMON_CASTLE_POSITION) < SAFE_ZONE_DEMON:
		return int(terrain_generator.T_COURTYARD)
	if center.distance_to(STARTER_CASTLE_POSITION) < SAFE_ZONE_CASTLE:
		return int(terrain_generator.T_COURTYARD)
	for camp_position: Vector2 in TRIAL_CAMPS.values():
		if center.distance_to(camp_position) < SAFE_ZONE_CAMP:
			return int(terrain_generator.T_CAMP)
	return terrain_generator.tile_id_at(tile_coord)

func _starter_river_tile(tile_coord: Vector2i) -> int:
	# 첫 모험에서 물길과 다리의 의미를 바로 배울 수 있도록 시작 성 동쪽에만
	# 짧은 고정 랜드마크를 둡니다. 이 구간 밖의 모든 지형은 WFC가 생성합니다.
	if not _river_wet(tile_coord):
		return -1
	if tile_coord.y == RIVER_BRIDGE_ROW and tile_coord.x >= RIVER_LEFT and tile_coord.x <= RIVER_RIGHT:
		return int(terrain_generator.T_BRIDGE)
	# 물가 타일은 WFC와 완전히 같은 규칙으로 고릅니다. 그래서 물길 끝과
	# 모서리가 호수와 같은 모양으로 닫힙니다.
	return int(terrain_generator.water_tile_for_pattern(
		not _river_wet(tile_coord + Vector2i(0, -1)),
		not _river_wet(tile_coord + Vector2i(1, 0)),
		not _river_wet(tile_coord + Vector2i(0, 1)),
		not _river_wet(tile_coord + Vector2i(-1, 0))
	))

func _river_wet(tile_coord: Vector2i) -> bool:
	if tile_coord.y < RIVER_TOP or tile_coord.y > RIVER_BOTTOM:
		return false
	return tile_coord.x >= RIVER_LEFT and tile_coord.x <= RIVER_RIGHT

func _is_water(point: Vector2) -> bool:
	var tile_id := _resolved_tile_id(_world_to_tile(point))
	return terrain_generator.is_water_tile(tile_id)

func _is_bridge(point: Vector2) -> bool:
	return terrain_generator.is_bridge_tile(_resolved_tile_id(_world_to_tile(point)))

func _features_near(point: Vector2, chunk_radius: int) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	var center_chunk := Vector2i(floori(point.x / FEATURE_CHUNK), floori(point.y / FEATURE_CHUNK))
	for cx in range(center_chunk.x - chunk_radius, center_chunk.x + chunk_radius + 1):
		for cy in range(center_chunk.y - chunk_radius, center_chunk.y + chunk_radius + 1):
			var feature := _feature_for_chunk(Vector2i(cx, cy))
			if not feature.is_empty():
				result.append(feature)
	return result

func _feature_for_chunk(chunk: Vector2i) -> Dictionary:
	if chunk == Vector2i(0, -1):
		return {"id":"castle_0_-1", "type":"castle", "position":Vector2(250.0, -250.0), "variant":0}
	var demon_chunk := Vector2i(floori(DEMON_CASTLE_POSITION.x / FEATURE_CHUNK), floori(DEMON_CASTLE_POSITION.y / FEATURE_CHUNK))
	if chunk == demon_chunk:
		return {"id":"demon_castle", "type":"demon_castle", "position":DEMON_CASTLE_POSITION, "variant":0}

	var value: int = absi(_hash(chunk.x, chunk.y, 97))
	var offset: Vector2 = Vector2(float((value >> 3) % 340 - 170), float((value >> 11) % 320 - 160))
	var position: Vector2 = Vector2(chunk.x * FEATURE_CHUNK + FEATURE_CHUNK * 0.5, chunk.y * FEATURE_CHUNK + FEATURE_CHUNK * 0.5) + offset
	if not is_walkable(position):
		# 호수가 조각이 아니라 제대로 된 덩어리가 됐으므로, 물 위에 놓인 목표를
		# 뭍으로 옮길 때 한 단계 반경으로는 부족할 수 있습니다. 결정적인 순서로
		# 반경을 넓혀 가며 찾습니다.
		var directions := [
			Vector2(1.0, 0.0), Vector2(-1.0, 0.0), Vector2(0.0, 1.0), Vector2(0.0, -1.0),
			Vector2(0.7, 0.7), Vector2(-0.7, 0.7), Vector2(0.7, -0.7), Vector2(-0.7, -0.7)
		]
		var relocated := false
		for reach in [230.0, 340.0, 450.0, 560.0]:
			for direction: Vector2 in directions:
				if is_walkable(position + direction * reach):
					position += direction * reach
					relocated = true
					break
			if relocated:
				break
	var feature_type: String = ""
	var roll: int = value % 100
	if roll < 9:
		feature_type = "castle"
	elif roll < 49:
		feature_type = "chest"
	elif roll < 76:
		feature_type = "grove"
	elif roll < 90:
		feature_type = "ruin"
	else:
		return {}
	return {
		"id":"%s_%d_%d" % [feature_type, chunk.x, chunk.y],
		"type":feature_type,
		"position":position,
		"variant":value % 4
	}

func _draw_castle(position: Vector2, demon: bool) -> void:
	var stone := Color("3b3045") if demon else GamePalette.STONE
	var dark := Color("1d1729") if demon else GamePalette.STONE_DARK
	var roof := GamePalette.RED.darkened(0.35) if demon else GamePalette.BLUE
	# Ground and road
	draw_rect(Rect2(position + Vector2(-104.0, 64.0), Vector2(208.0, 44.0)), GamePalette.DIRT_DARK, true)
	# Main keep
	draw_rect(Rect2(position + Vector2(-72.0, -60.0), Vector2(144.0, 126.0)), dark, true)
	draw_rect(Rect2(position + Vector2(-64.0, -52.0), Vector2(128.0, 112.0)), stone, true)
	# Towers and crenellations
	for tower_x in [-90.0, 66.0]:
		draw_rect(Rect2(position + Vector2(tower_x, -48.0), Vector2(48.0, 112.0)), dark, true)
		draw_rect(Rect2(position + Vector2(tower_x + 5.0, -44.0), Vector2(38.0, 104.0)), stone, true)
		for tooth in 4:
			draw_rect(Rect2(position + Vector2(tower_x + tooth * 13.0, -58.0), Vector2(9.0, 15.0)), stone, true)
	for tooth in 7:
		draw_rect(Rect2(position + Vector2(-64.0 + tooth * 21.0, -66.0), Vector2(13.0, 17.0)), stone, true)
	# Roof, banner, windows, gate
	var roof_points := PackedVector2Array([position + Vector2(-48.0,-52.0), position + Vector2(0.0,-100.0), position + Vector2(48.0,-52.0)])
	draw_colored_polygon(roof_points, roof)
	draw_rect(Rect2(position + Vector2(-4.0, -122.0), Vector2(8.0, 30.0)), dark, true)
	draw_rect(Rect2(position + Vector2(4.0, -119.0), Vector2(28.0, 14.0)), GamePalette.RED if demon else GamePalette.YELLOW, true)
	draw_rect(Rect2(position + Vector2(-16.0, 22.0), Vector2(32.0, 44.0)), Color("1a1423") if demon else GamePalette.WOOD, true)
	draw_rect(Rect2(position + Vector2(-40.0, -18.0), Vector2(14.0, 22.0)), GamePalette.RED if demon else GamePalette.CYAN, true)
	draw_rect(Rect2(position + Vector2(26.0, -18.0), Vector2(14.0, 22.0)), GamePalette.RED if demon else GamePalette.CYAN, true)
	if not demon:
		# Three tiny random-service NPC silhouettes at the gate.
		for index in 3:
			var npc_pos := position + Vector2(-42.0 + index * 42.0, 92.0)
			draw_rect(Rect2(npc_pos - Vector2(6.0, 13.0), Vector2(12.0, 18.0)), [GamePalette.ORANGE, GamePalette.GREEN, GamePalette.MAGENTA][index], true)
			draw_rect(Rect2(npc_pos - Vector2(4.0, 20.0), Vector2(8.0, 8.0)), GamePalette.TEXT, true)
	else:
		draw_circle(position + Vector2(0.0, -18.0), 9.0, GamePalette.RED)

func _draw_chest(position: Vector2) -> void:
	draw_rect(Rect2(position + Vector2(-18.0, -11.0), Vector2(36.0, 25.0)), GamePalette.WOOD, true)
	draw_rect(Rect2(position + Vector2(-18.0, -14.0), Vector2(36.0, 9.0)), GamePalette.WOOD_LIGHT, true)
	draw_rect(Rect2(position + Vector2(-3.0, -8.0), Vector2(7.0, 16.0)), GamePalette.YELLOW, true)
	draw_rect(Rect2(position + Vector2(-20.0, -16.0), Vector2(40.0, 31.0)), GamePalette.DIRT_DARK, false, 3.0)

func _draw_grove(position: Vector2, variant: int) -> void:
	for index in 5:
		var angle := TAU * float(index) / 5.0 + variant * 0.3
		var tree := position + Vector2.from_angle(angle) * (24.0 + index % 2 * 18.0)
		draw_rect(Rect2(tree + Vector2(-4.0, 5.0), Vector2(8.0, 18.0)), GamePalette.WOOD, true)
		draw_rect(Rect2(tree + Vector2(-13.0, -15.0), Vector2(26.0, 24.0)), GamePalette.GRASS_DARK.darkened(0.15), true)
		draw_rect(Rect2(tree + Vector2(-8.0, -20.0), Vector2(16.0, 10.0)), GamePalette.GRASS_LIGHT.darkened(0.12), true)

func _draw_trial_camp(position: Vector2, direction: String, cleared: bool) -> void:
	var colors := {"north":GamePalette.CYAN, "east":GamePalette.YELLOW, "south":GamePalette.RED, "west":GamePalette.MAGENTA}
	var camp_color: Color = colors.get(direction, GamePalette.TEXT)
	draw_circle(position, 122.0, Color(camp_color, 0.055 if not cleared else 0.025))
	draw_arc(position, 122.0, 0.0, TAU, 32, Color(camp_color, 0.55), 4.0)
	if cleared:
		draw_arc(position, 58.0, 0.0, TAU, 16, Color(camp_color, 0.7), 4.0)
		draw_line(position + Vector2(-25.0, 0.0), position + Vector2(-5.0, 22.0), camp_color, 8.0)
		draw_line(position + Vector2(-5.0, 22.0), position + Vector2(34.0, -25.0), camp_color, 8.0)
		return
	# Four tents and one tall cardinal banner make the camp readable at a glance.
	for index in 4:
		var tent_center := position + Vector2.from_angle(TAU * float(index) / 4.0 + PI * 0.25) * 62.0
		var tent := PackedVector2Array([tent_center+Vector2(-22,14),tent_center+Vector2(0,-20),tent_center+Vector2(22,14)])
		draw_colored_polygon(tent, GamePalette.DIRT_DARK)
		draw_polyline(PackedVector2Array(Array(tent)+[tent[0]]), camp_color.darkened(0.15), 3.0)
	draw_rect(Rect2(position + Vector2(-4.0, -96.0), Vector2(8.0, 100.0)), GamePalette.WOOD, true)
	draw_rect(Rect2(position + Vector2(4.0, -92.0), Vector2(54.0, 28.0)), camp_color, true)
	draw_rect(Rect2(position + Vector2(12.0, -85.0), Vector2(9.0, 14.0)), Color.WHITE, true)

func _hash(x: int, y: int, salt: int) -> int:
	var value := x * 374761393 + y * 668265263 + salt * 1442695041
	value = (value ^ (value >> 13)) * 1274126177
	return value ^ (value >> 16)
